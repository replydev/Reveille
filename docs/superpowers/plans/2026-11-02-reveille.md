# Reveille Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Reveille, a sideloaded Android 12+ alarm app with one-off date alarms, policies of recurring rules, a next-alarm widget, and defense-in-depth against missed wake-ups (including on HyperOS).

**Architecture:** A pure-Kotlin `:engine` module computes upcoming occurrences from a schedule snapshot and decides what to arm (`ArmPlanner`). The `:app` module stores the schedule in Room (device-protected storage), arms exactly one primary alarm plus one backup with `AlarmManager.setAlarmClock()`, rings through a `systemExempted` foreground service with a full-screen activity, and re-arms on every change, fire, boot, time change and from a periodic watchdog. A Glance widget renders the armed alarm.

**Tech Stack:** Kotlin 2.4.21, AGP 9.5.0 (built-in Kotlin), Gradle 9.8.2, Jetpack Compose (BOM 2026.09.00, Material 3), Room 2.8.5 + KSP 2.3.13, DataStore 1.3.0, WorkManager 2.12.0, Glance 1.2.0, JUnit 4, Robolectric 4.17. minSdk 31, compileSdk/targetSdk 37.

**Spec:** `docs/superpowers/specs/2026-11-02-reveille-design.md`

- [ ] **Step 3: Write `Model.kt`**

```kotlin
package com.replydev.reveille.engine

import java.time.DayOfWeek
import java.time.Instant
import java.time.LocalDate
import java.time.LocalTime
import java.time.ZonedDateTime

/** A recurring alarm time inside a policy. [anchorDate] picks which weeks count when [intervalWeeks] > 1. */
data class Rule(
    val id: Long,
    val time: LocalTime,
    val days: Set<DayOfWeek>,
    val intervalWeeks: Int = 1,
    val anchorDate: LocalDate = LocalDate.of(2026, 1, 5),
    val label: String? = null,
)

/** Inclusive date range during which a policy does not ring. */
data class Pause(val from: LocalDate, val to: LocalDate) {
    fun contains(date: LocalDate): Boolean = !date.isBefore(from) && !date.isAfter(to)
}

data class Policy(
    val id: Long,
    val name: String,
    val enabled: Boolean = true,
    val startDate: LocalDate? = null,
    val endDate: LocalDate? = null,
    val rules: List<Rule> = emptyList(),
    val pauses: List<Pause> = emptyList(),
)

enum class OneOffMode { REPLACE_DAY, ADDITIVE }

data class OneOff(
    val id: Long,
    val date: LocalDate,
    val time: LocalTime,
    val label: String? = null,
    val mode: OneOffMode = OneOffMode.REPLACE_DAY,
)

data class Skip(val policyId: Long, val date: LocalDate)

data class ScheduleSnapshot(
    val policies: List<Policy> = emptyList(),
    val oneOffs: List<OneOff> = emptyList(),
    val skips: List<Skip> = emptyList(),
)

/** What produced an occurrence. [key] is stable and stored in skips, logs and armed state. */
sealed interface Source {
    val key: String

    data class PolicyRule(val policyId: Long, val ruleId: Long, val date: LocalDate) : Source {
        override val key: String get() = "P:$policyId:$ruleId:$date"
    }

    data class OneOffAlarm(val oneOffId: Long) : Source {
        override val key: String get() = "O:$oneOffId"
    }

    companion object {
        fun parse(key: String): Source? {
            val parts = key.split(":")
            return when {
                parts.size == 4 && parts[0] == "P" -> {
                    val policyId = parts[1].toLongOrNull() ?: return null
                    val ruleId = parts[2].toLongOrNull() ?: return null
                    val date = runCatching { LocalDate.parse(parts[3]) }.getOrNull() ?: return null
                    PolicyRule(policyId, ruleId, date)
                }
                parts.size == 2 && parts[0] == "O" -> parts[1].toLongOrNull()?.let(::OneOffAlarm)
                else -> null
            }
        }
    }

/** One moment at which the phone should ring. Alarms at the same instant are merged into one occurrence. */
data class Occurrence(val at: ZonedDateTime, val label: String, val sources: List<Source>) {
    val key: String get() = sources.first().key
}

/** The alarm currently registered with AlarmManager (or about to be). */
data class Armed(
    val key: String,
    val at: Instant,
    val label: String,
    val sourceKeys: List<String> = emptyList(),
) {
    val isSnooze: Boolean get() = key.startsWith(SNOOZE_PREFIX)

    fun encode(): String =
        listOf(key, at.toEpochMilli().toString(), label.replace(SEPARATOR, " "), sourceKeys.joinToString(","))
            .joinToString(SEPARATOR)

    companion object {
        const val SNOOZE_PREFIX = "S:"
        private const val SEPARATOR = "\u001F"

        fun snooze(at: Instant, label: String): Armed = Armed("$SNOOZE_PREFIX${at.toEpochMilli()}", at, label)

        fun decode(value: String): Armed? {
            val parts = value.split(SEPARATOR)
            if (parts.size != 4) return null
            val millis = parts[1].toLongOrNull() ?: return null
            return Armed(parts[0], Instant.ofEpochMilli(millis), parts[2], parts[3].split(",").filter { it.isNotEmpty() })
        }
    }
}

fun Occurrence.toArmed(): Armed = Armed(key, at.toInstant(), label, sources.map { it.key })
```

```kotlin
package com.replydev.reveille.engine

import java.time.DayOfWeek
import java.time.Instant
import java.time.LocalDate
import java.time.LocalTime
import java.time.ZoneId
import java.time.ZonedDateTime
import java.time.temporal.ChronoUnit
import java.time.temporal.TemporalAdjusters

/** An alarm on a specific day, before it is resolved to an instant. */
data class DayAlarm(val time: LocalTime, val label: String, val source: Source)

object ScheduleEngine {
    const val DEFAULT_HORIZON_DAYS = 400
    const val DEFAULT_ONE_OFF_LABEL = "Alarm"

    /** Occurrences strictly after [from], in order, looking at most [horizonDays] days past [from]'s date. */
    fun upcoming(
        snapshot: ScheduleSnapshot,
        from: Instant,
        zone: ZoneId,
        limit: Int,
        horizonDays: Int = DEFAULT_HORIZON_DAYS,
    ): List<Occurrence> {
        if (limit <= 0) return emptyList()
        val firstDay = from.atZone(zone).toLocalDate()
        val result = mutableListOf<Occurrence>()
        for (offset in 0..horizonDays) {
            occurrencesOn(snapshot, firstDay.plusDays(offset.toLong()), zone)
                .filterTo(result) { it.at.toInstant().isAfter(from) }
            if (result.size >= limit) break
        }
        return result.take(limit)
    }

    fun occurrencesOn(snapshot: ScheduleSnapshot, day: LocalDate, zone: ZoneId): List<Occurrence> {
        val oneOffs = snapshot.oneOffs.filter { it.date == day }
        val replacesDay = oneOffs.any { it.mode == OneOffMode.REPLACE_DAY }
        val policyAlarms = if (replacesDay) emptyList() else policyAlarmsOn(snapshot, day)
        val oneOffAlarms = oneOffs.map { DayAlarm(it.time, it.label ?: DEFAULT_ONE_OFF_LABEL, Source.OneOffAlarm(it.id)) }
        return (policyAlarms + oneOffAlarms)
            .groupBy { ZonedDateTime.of(day, it.time, zone) }
            .entries
            .sortedBy { it.key.toInstant() }
            .map { (at, group) ->
                Occurrence(
                    at = at,
                    label = group.map { it.label }.distinct().joinToString(" · "),
                    sources = group.map { it.source },
                )
            }
    }

    /** Policy alarms on [day], ignoring one-offs. The one-off editor uses this to show what it replaces. */
    fun policyAlarmsOn(snapshot: ScheduleSnapshot, day: LocalDate): List<DayAlarm> =
        snapshot.policies
            .filter { isActiveOn(it, day, snapshot.skips) }
            .flatMap { policy ->
                policy.rules
                    .filter { ruleMatches(it, day) }
                    .map { rule -> DayAlarm(rule.time, rule.label ?: policy.name, Source.PolicyRule(policy.id, rule.id, day)) }
            }
            .sortedBy { it.time }

    fun ruleMatches(rule: Rule, day: LocalDate): Boolean {
        if (day.dayOfWeek !in rule.days) return false
        val interval = rule.intervalWeeks.coerceAtLeast(1)
        if (interval == 1) return true
        val weeks = ChronoUnit.WEEKS.between(mondayOf(rule.anchorDate), mondayOf(day))
        return Math.floorMod(weeks, interval.toLong()) == 0L
    }

    private fun isActiveOn(policy: Policy, day: LocalDate, skips: List<Skip>): Boolean {
        val start = policy.startDate
        val end = policy.endDate
        return policy.enabled &&
            (start == null || !day.isBefore(start)) &&
            (end == null || !day.isAfter(end)) &&
            policy.pauses.none { it.contains(day) } &&
            skips.none { it.policyId == policy.id && it.date == day }
    }

    private fun mondayOf(date: LocalDate): LocalDate = date.with(TemporalAdjusters.previousOrSame(DayOfWeek.MONDAY))
}
```

```kotlin
package com.replydev.reveille.engine

import java.time.Duration
import java.time.Instant

/** [armed] is what to register, [triggerAt] when AlarmManager should fire it, [missed] an alarm to report as missed. */
data class ArmDecision(val armed: Armed?, val triggerAt: Instant?, val missed: Armed?)

object ArmPlanner {
    /** An alarm that should have rung up to this long ago still rings (e.g. after a reboot). */
    val LATE_RING_WINDOW: Duration = Duration.ofMinutes(30)

    /** The backup alarm fires this long after the primary. */
    val BACKUP_DELAY: Duration = Duration.ofMinutes(2)

    /** Catch-up alarms are re-armed this far in the future so they go through AlarmManager. */
    val CATCH_UP_DELAY: Duration = Duration.ofSeconds(1)

    fun decide(previous: Armed?, previousHandled: Boolean, next: Armed?, snooze: Armed?, now: Instant): ArmDecision {
        var missed: Armed? = null
        if (previous != null && !previousHandled && !previous.at.isAfter(now)) {
            if (Duration.between(previous.at, now) <= LATE_RING_WINDOW) {
                return ArmDecision(previous, now.plus(CATCH_UP_DELAY), null)
            }
            missed = previous
        }
        val upcoming = listOfNotNull(next, snooze?.takeIf { it.at.isAfter(now) }).minByOrNull { it.at }
        return ArmDecision(upcoming, upcoming?.at, missed)
    }
}
```

```kotlin
import java.time.Instant

object Timeline {
    data class Entry(val occurrence: Occurrence, val skipped: Boolean)

    /** Occurrences from [from] through the end of the [days]-th day, including skipped ones (flagged). */
    fun build(snapshot: ScheduleSnapshot, from: Instant, zone: ZoneId, days: Int): List<Entry> {
        val skips = snapshot.skips.toSet()
        return ScheduleEngine
            .upcoming(snapshot.copy(skips = emptyList()), from, zone, limit = Int.MAX_VALUE, horizonDays = days - 1)
            .map { occurrence ->
                val skipped = occurrence.sources.all { it is Source.PolicyRule && Skip(it.policyId, it.date) in skips }
                Entry(occurrence, skipped)
            }
    }
}
```

```kotlin
package com.replydev.reveille.engine

import java.time.DayOfWeek
import java.time.Duration
import java.time.Instant
import java.time.LocalDate
import java.time.LocalTime
import java.time.ZoneId
import java.time.format.DateTimeFormatter
import java.time.format.TextStyle
import java.util.Locale

object TimeText {
    private val timeFormat = DateTimeFormatter.ofPattern("HH:mm")
    private val weekdayFormat = DateTimeFormatter.ofPattern("EEE", Locale.ENGLISH)
    private val shortDateFormat = DateTimeFormatter.ofPattern("EEE d MMM", Locale.ENGLISH)
    private val fullDateFormat = DateTimeFormatter.ofPattern("d MMM yyyy", Locale.ENGLISH)
    private val workWeek = setOf(DayOfWeek.MONDAY, DayOfWeek.TUESDAY, DayOfWeek.WEDNESDAY, DayOfWeek.THURSDAY, DayOfWeek.FRIDAY)

    fun time(value: LocalTime): String = value.format(timeFormat)

    fun date(value: LocalDate): String = value.format(fullDateFormat)

    fun dayHeader(date: LocalDate, today: LocalDate): String = when (date) {
        today -> "Today"
        today.plusDays(1) -> "Tomorrow"
        else -> date.format(shortDateFormat)
    }

    fun dayAndTime(at: Instant, now: Instant, zone: ZoneId): String {
        val local = at.atZone(zone)
        val today = now.atZone(zone).toLocalDate()
        val date = local.toLocalDate()
        val day = when {
            date == today -> "Today"
            date == today.plusDays(1) -> "Tomorrow"
            date.isBefore(today.plusDays(7)) -> date.format(weekdayFormat)
            else -> date.format(shortDateFormat)
        }
        return "$day ${time(local.toLocalTime())}"
    }

    fun relative(at: Instant, now: Instant): String {
        val minutes = Duration.between(now, at).toMinutes()
        return when {
            minutes < 1 -> "now"
            minutes < 60 -> "in $minutes min"
            minutes < 48 * 60 -> "in ${(minutes + 30) / 60} h"
            else -> "in ${(minutes + 12 * 60) / (24 * 60)} d"
        }
    }

    fun days(days: Set<DayOfWeek>): String = when {
        days.size == 7 -> "Every day"
        days == workWeek -> "Mon–Fri"
        else -> DayOfWeek.entries.filter { it in days }.joinToString(" ") { it.getDisplayName(TextStyle.SHORT, Locale.ENGLISH) }
    }
}
```

```kotlin
    suspend fun deletePolicy(id: Long)

    @Query("UPDATE policy SET enabled = :enabled WHERE id = :id")
    suspend fun setPolicyEnabled(id: Long, enabled: Boolean)

    @Query("DELETE FROM rule WHERE policyId = :policyId")
    suspend fun deleteRules(policyId: Long)

    @Insert
    suspend fun insertRules(rules: List<RuleEntity>)

    @Query("DELETE FROM pause WHERE policyId = :policyId")
    suspend fun deletePauses(policyId: Long)

    @Insert
    suspend fun insertPauses(pauses: List<PauseEntity>)

    /** Inserts or updates [policy] and replaces all its rules and pauses. Returns the policy id. */
    @Transaction
    suspend fun savePolicy(policy: PolicyEntity, rules: List<RuleEntity>, pauses: List<PauseEntity>): Long {
        val id = if (policy.id == 0L) insertPolicy(policy) else policy.id.also { updatePolicy(policy) }
        deleteRules(id)
        insertRules(rules.map { it.copy(id = 0, policyId = id) })
        deletePauses(id)
        insertPauses(pauses.map { it.copy(id = 0, policyId = id) })
        return id
    }

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun saveOneOff(oneOff: OneOffEntity): Long

    @Query("DELETE FROM one_off WHERE id = :id")
    suspend fun deleteOneOff(id: Long)

    @Insert(onConflict = OnConflictStrategy.IGNORE)
    suspend fun insertSkip(skip: SkipEntity)

    @Query("DELETE FROM skip WHERE policyId = :policyId AND date = :date")
    suspend fun deleteSkip(policyId: Long, date: LocalDate)

    @Query("DELETE FROM one_off WHERE date < :before")
    suspend fun deleteOneOffsBefore(before: LocalDate)

    @Query("DELETE FROM skip WHERE date < :before")
    suspend fun deleteSkipsBefore(before: LocalDate)
```

```kotlin
package com.replydev.reveille.data

import com.replydev.reveille.engine.ScheduleSnapshot
import com.replydev.reveille.engine.Source
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.combine
import java.time.LocalDate

/** All schedule writes go through here; each write calls [onChanged] so the scheduler re-arms. */
class ScheduleRepository(private val dao: ScheduleDao, private val onChanged: suspend () -> Unit) {
    val policies: Flow<List<PolicyWithChildren>> = dao.observePolicies()

    val snapshots: Flow<ScheduleSnapshot> =
        combine(dao.observePolicies(), dao.observeOneOffs(), dao.observeSkips()) { policies, oneOffs, skips ->
            ScheduleSnapshot(policies.map { it.toModel() }, oneOffs.map { it.toModel() }, skips.map { it.toModel() })
        }

    suspend fun snapshot(): ScheduleSnapshot =
        ScheduleSnapshot(dao.policies().map { it.toModel() }, dao.oneOffs().map { it.toModel() }, dao.skips().map { it.toModel() })

    suspend fun policy(id: Long): PolicyWithChildren? = dao.policy(id)

    suspend fun oneOff(id: Long): OneOffEntity? = dao.oneOff(id)

    suspend fun savePolicy(policy: PolicyEntity, rules: List<RuleEntity>, pauses: List<PauseEntity>): Long =
        dao.savePolicy(policy, rules, pauses).also { onChanged() }

    suspend fun deletePolicy(id: Long) {
        dao.deletePolicy(id)
        onChanged()
    }

    suspend fun setPolicyEnabled(id: Long, enabled: Boolean) {
        dao.setPolicyEnabled(id, enabled)
        onChanged()
    }

    suspend fun saveOneOff(oneOff: OneOffEntity): Long = dao.saveOneOff(oneOff).also { onChanged() }

    suspend fun deleteOneOff(id: Long) {
        dao.deleteOneOff(id)
        onChanged()
    }

    suspend fun unskip(policyId: Long, date: LocalDate) {
        dao.deleteSkip(policyId, date)
        onChanged()
    }

    /** Skips policy occurrences and deletes one-offs referenced by [sourceKeys]. */
    suspend fun skipOccurrence(sourceKeys: List<String>) {
        sourceKeys.mapNotNull(Source::parse).forEach { source ->
            when (source) {
                is Source.PolicyRule -> dao.insertSkip(SkipEntity(source.policyId, source.date))
                is Source.OneOffAlarm -> dao.deleteOneOff(source.oneOffId)
            }
        }
        onChanged()
    }

    /** Deletes one-offs and skips dated before [today]. Does not call [onChanged]: it runs inside a re-arm. */
    suspend fun cleanup(today: LocalDate) {
        dao.deleteOneOffsBefore(today)
        dao.deleteSkipsBefore(today)
    }
}
```

```kotlin
package com.replydev.reveille.data
```

```kotlin
import androidx.datastore.core.DataStore
import androidx.datastore.preferences.core.Preferences
```

```kotlin
data class RingSettings(
    val ringtoneUri: String? = null,
    val vibrate: Boolean = true,
    val snoozeMinutes: Int = 9,
    val maxRingMinutes: Int = 10,
    val volumeRamp: Boolean = true,
)

class SettingsStore(private val store: DataStore<Preferences>) {
    val settings: Flow<RingSettings> = store.data.map(::read)

    /** Ids of manual reliability checks the user has confirmed. */
    val confirmedChecks: Flow<Set<String>> = store.data.map { it[CONFIRMED] ?: emptySet() }

    suspend fun current(): RingSettings = settings.first()

    suspend fun update(transform: (RingSettings) -> RingSettings) {
        store.edit { prefs ->
            val next = transform(read(prefs))
            if (next.ringtoneUri == null) prefs.remove(RINGTONE) else prefs[RINGTONE] = next.ringtoneUri
            prefs[VIBRATE] = next.vibrate
            prefs[SNOOZE_MINUTES] = next.snoozeMinutes
            prefs[MAX_RING_MINUTES] = next.maxRingMinutes
            prefs[VOLUME_RAMP] = next.volumeRamp
        }
    }

    suspend fun setConfirmed(id: String, confirmed: Boolean) {
        store.edit { prefs ->
            val current = prefs[CONFIRMED] ?: emptySet()
            prefs[CONFIRMED] = if (confirmed) current + id else current - id
        }
    }

    private fun read(prefs: Preferences): RingSettings {
        val defaults = RingSettings()
        return RingSettings(
            ringtoneUri = prefs[RINGTONE],
            vibrate = prefs[VIBRATE] ?: defaults.vibrate,
            snoozeMinutes = prefs[SNOOZE_MINUTES] ?: defaults.snoozeMinutes,
            maxRingMinutes = prefs[MAX_RING_MINUTES] ?: defaults.maxRingMinutes,
            volumeRamp = prefs[VOLUME_RAMP] ?: defaults.volumeRamp,
        )
    }

    private companion object {
        val RINGTONE = stringPreferencesKey("ringtone_uri")
        val VIBRATE = booleanPreferencesKey("vibrate")
        val SNOOZE_MINUTES = intPreferencesKey("snooze_minutes")
        val MAX_RING_MINUTES = intPreferencesKey("max_ring_minutes")
        val VOLUME_RAMP = booleanPreferencesKey("volume_ramp")

    }
```

How it fits together:
1. `AlarmScheduler.rearm()` (serialized by a mutex) asks the engine for the next occurrence, runs `ArmPlanner`, and registers a **primary** and a **backup** (`+2 min`) alarm with `setAlarmClock()`. Both carry the same encoded `Armed`.
2. `AlarmReceiver` gets either. It **claims** the occurrence by inserting into the fire log (unique index). Only the first claim starts `RingingService`; duplicates (backup after a successful primary, catch-up racing a real fire) are ignored. Then it re-arms.
3. `RingingService` is a `systemExempted` foreground service. It plays the alarm sound with `USAGE_ALARM`, falling back to a bundled WAV (works before first unlock, when media content providers are unavailable). It vibrates and posts a full-screen notification that opens `RingingActivity`. Sound never depends on the activity.
4. Snooze stores an `Armed.snooze(...)` in `SchedulerState` and re-arms. After `maxRingMinutes` without interaction it auto-snoozes. The 4th timeout stops, logs the alarm as missed, and notifies.

```python
import math
import struct
import sys
import wave

RATE = 22050
frames = bytearray()
for i in range(RATE):
    t = i / RATE
    on = (t % 0.5) < 0.25
    value = int(32767 * 0.8 * math.sin(2 * math.pi * 880 * t)) if on else 0
    frames += struct.pack("<h", value)

with wave.open(sys.argv[1], "wb") as out:
    out.setnchannels(1)
    out.setsampwidth(2)
    out.setframerate(RATE)
    out.writeframes(bytes(frames))
```

```kotlin
package com.replydev.reveille.alarm

import android.app.AlarmManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import android.util.Log
import com.replydev.reveille.MainActivity
import com.replydev.reveille.data.FireLogDao
import com.replydev.reveille.data.FireLogEntity
import com.replydev.reveille.data.FireOutcome
import com.replydev.reveille.data.ScheduleRepository
import com.replydev.reveille.data.SchedulerState
import com.replydev.reveille.engine.ArmPlanner
import com.replydev.reveille.engine.Armed
import kotlinx.coroutines.sync.Mutex
import kotlinx.coroutines.sync.withLock
import java.time.Instant
import java.time.LocalDate
import java.time.ZoneId

class AlarmScheduler(
    private val context: Context,
    private val repository: ScheduleRepository,
    private val fireLog: FireLogDao,
    private val state: SchedulerState,
    private val notifier: Notifier,
    private val onArmedChanged: suspend () -> Unit,
    private val clock: () -> Instant = Instant::now,
) {
    private val alarmManager = context.getSystemService(AlarmManager::class.java)
    private val mutex = Mutex()

    /** Recomputes and registers the next alarm. Safe to call from anywhere, any number of times. */
    suspend fun rearm(reason: String): Unit = mutex.withLock {
        val now = clock()
        val zone = ZoneId.systemDefault()
        // Keep two extra days of one-offs/skips so overdue alarms from just before midnight can still be caught up.
        repository.cleanup(LocalDate.ofInstant(now, zone).minusDays(2))
        val previous = state.armed()
        val snooze = state.snooze()
        val since = listOfNotNull(previous?.at, snooze?.at, now).min()
        val handled = fireLog.handledSince(since.toEpochMilli()).mapTo(HashSet()) { it.occurrenceKey to it.scheduledAt }
        val decision = ArmPlanner.decide(repository.snapshot(), zone, now, previous, snooze) { (it.key to it.at.toEpochMilli()) in handled }

        decision.missed.forEach { recordMissed(it) }
        if (decision.missed.any { it.isSnooze }) state.setSnooze(null)
        val armed = decision.armed
        val trigger = decision.triggerAt
        if (armed != null && trigger != null) arm(armed, trigger) else cancel()
        state.setArmed(armed)
        Log.i(TAG, "rearm($reason): armed=${armed?.key} trigger=$trigger missed=${decision.missed.map { it.key }}")
        onArmedChanged()
    }

    /** Returns true only for the first delivery of [armed]; later deliveries (backup, catch-up) return false. */
    suspend fun claim(armed: Armed): Boolean =
        fireLog.insert(log(armed, FireOutcome.FIRED, firedAt = clock().toEpochMilli())) != -1L

    suspend fun recordOutcome(armed: Armed, outcome: FireOutcome) =
        fireLog.setOutcome(armed.key, armed.at.toEpochMilli(), outcome)

    /** Skips the currently armed alarm (widget "Skip" button). */
    suspend fun skipNext() {
        val armed = state.armed() ?: return
        // A past (catch-up) alarm must be marked handled, otherwise the planner would ring it again.
        if (!armed.at.isAfter(clock())) fireLog.insert(log(armed, FireOutcome.SKIPPED, firedAt = null))
        if (armed.isSnooze || armed.sourceKeys.isEmpty()) {
            state.setSnooze(null)
            rearm("skipped next")
        } else {
            repository.skipOccurrence(armed.sourceKeys)
        }
    }

    private suspend fun recordMissed(missed: Armed) {
        fireLog.insert(log(missed, FireOutcome.MISSED, firedAt = null))
        notifier.showMissed(missed)
    }

    private fun log(armed: Armed, outcome: FireOutcome, firedAt: Long?) = FireLogEntity(
        occurrenceKey = armed.key,
        label = armed.label,
        scheduledAt = armed.at.toEpochMilli(),
        firedAt = firedAt,
        outcome = outcome,
    )

    private fun arm(armed: Armed, trigger: Instant) {
        try {
            alarmManager.setAlarmClock(AlarmManager.AlarmClockInfo(trigger.toEpochMilli(), showIntent()), fireIntent(armed, REQUEST_PRIMARY))
            alarmManager.setAlarmClock(
                AlarmManager.AlarmClockInfo(trigger.plus(ArmPlanner.BACKUP_DELAY).toEpochMilli(), showIntent()),
                fireIntent(armed, REQUEST_BACKUP),
            )
        } catch (e: SecurityException) {
            Log.e(TAG, "Exact alarms not permitted", e)
            notifier.showSchedulingFailed("Reveille is not allowed to schedule exact alarms. Open Reveille → Reliability.")
        }
    }

    private fun cancel() {
        alarmManager.cancel(fireIntent(null, REQUEST_PRIMARY))
        alarmManager.cancel(fireIntent(null, REQUEST_BACKUP))
    }

    private fun showIntent(): PendingIntent = PendingIntent.getActivity(
        context,
        REQUEST_SHOW,
        Intent(context, MainActivity::class.java),
        PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT,
    )

    private fun fireIntent(armed: Armed?, requestCode: Int): PendingIntent {
        val intent = Intent(context, AlarmReceiver::class.java)
            .setAction(ACTION_FIRE)
            .putExtra(EXTRA_BACKUP, requestCode == REQUEST_BACKUP)
        if (armed != null) intent.putExtra(EXTRA_ARMED, armed.encode())
        return PendingIntent.getBroadcast(context, requestCode, intent, PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT)
    }

    companion object {
        const val ACTION_FIRE = "com.replydev.reveille.FIRE"
        const val EXTRA_ARMED = "armed"
        const val EXTRA_BACKUP = "backup"
        private const val REQUEST_PRIMARY = 100
        private const val REQUEST_BACKUP = 101
        private const val REQUEST_SHOW = 102
        private const val TAG = "Reveille"
    }
}
```

```kotlin
    override fun onReceive(context: Context, intent: Intent) {
        if (intent.action != AlarmScheduler.ACTION_FIRE) return
        val armed = intent.getStringExtra(AlarmScheduler.EXTRA_ARMED)?.let(Armed::decode) ?: return
        val backup = intent.getBooleanExtra(AlarmScheduler.EXTRA_BACKUP, false)
        val container = context.container
        val pending = goAsync()
        CoroutineScope(SupervisorJob() + Dispatchers.IO).launch {
            try {
                if (container.scheduler.claim(armed)) {
                    if (armed.isSnooze) container.schedulerState.setSnooze(null)
                    RingingService.start(context, armed, backup)
                }
                container.scheduler.rearm(if (backup) "backup fired" else "alarm fired")
            } finally {
                pending.finish()
            }
        }
    }
}
```

```kotlin
import kotlin.time.Duration.Companion.minutes
import kotlin.time.Duration.Companion.seconds

data class Ringing(val armed: Armed, val backup: Boolean) {
    val title: String get() = if (backup) "${armed.label} (backup)" else armed.label
}

class RingingService : Service() {
    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.Main.immediate)
    private var player: MediaPlayer? = null
    private var rampJob: Job? = null
    private var timeoutJob: Job? = null
    private var wakeLock: PowerManager.WakeLock? = null
    private val vibrator: Vibrator by lazy { getSystemService(VibratorManager::class.java).defaultVibrator }

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_RING -> {
                // Fail loud: if the extra is somehow unreadable, ring anyway with a generic label.
                val armed = intent.getStringExtra(EXTRA_ARMED)?.let(Armed::decode) ?: Armed("unknown", Instant.now(), "Alarm")
                ring(Ringing(armed, intent.getBooleanExtra(EXTRA_BACKUP, false)))
            }
            ACTION_SNOOZE -> snooze(auto = false)
            ACTION_DISMISS -> dismiss()
            else -> if (current.value == null) stopSelf()
        }
        return START_NOT_STICKY
    }

    private fun ring(ringing: Ringing) {
        current.value = ringing
        val notification = container.notifier.ringingNotification(ringing)
        try {
            startForeground(Notifier.RINGING_ID, notification, ServiceInfo.FOREGROUND_SERVICE_TYPE_SYSTEM_EXEMPTED)
        } catch (e: Exception) {
            Log.e(TAG, "startForeground failed; ringing without foreground status", e)
            container.notifier.postRinging(ringing)
        }
        stopOutput()
        timeoutJob?.cancel()
        acquireWakeLock()
        scope.launch {
            val settings = container.settings.current()
            startSound(settings)
            if (settings.vibrate) startVibration()
            timeoutJob = launch {
                delay(settings.maxRingMinutes.minutes)
                snooze(auto = true)
            }
        }
    }

    private fun snooze(auto: Boolean) {
        val ringing = current.value ?: return stopRinging()
        stopOutput()
        scope.launch {
            val c = container
            val count = if (auto) c.schedulerState.autoSnoozeCount() + 1 else 0
            if (count > MAX_AUTO_SNOOZES) {
                c.schedulerState.setAutoSnoozeCount(0)
                c.scheduler.recordOutcome(ringing.armed, FireOutcome.MISSED)
                c.notifier.showMissed(ringing.armed)
            } else {
                c.schedulerState.setAutoSnoozeCount(count)
                val at = Instant.now().plus(Duration.ofMinutes(c.settings.current().snoozeMinutes.toLong()))
                c.schedulerState.setSnooze(Armed.snooze(at, ringing.armed.label))
                c.scheduler.recordOutcome(ringing.armed, if (auto) FireOutcome.AUTO_SNOOZED else FireOutcome.SNOOZED)
                c.scheduler.rearm("snoozed")
            }
            stopRinging()
        }
    }

    private fun dismiss() {
        val ringing = current.value ?: return stopRinging()
        stopOutput()
        scope.launch {
            container.schedulerState.setAutoSnoozeCount(0)
            container.scheduler.recordOutcome(ringing.armed, FireOutcome.DISMISSED)
            stopRinging()
        }
    }

    private fun startSound(settings: RingSettings) {
        val attributes = AudioAttributes.Builder()
            .setUsage(AudioAttributes.USAGE_ALARM)
            .setContentType(AudioAttributes.CONTENT_TYPE_SONIFICATION)
            .build()
        val player = MediaPlayer()
        val chosen = runCatching {
            val uri = settings.ringtoneUri?.let(Uri::parse)
                ?: RingtoneManager.getActualDefaultRingtoneUri(this, RingtoneManager.TYPE_ALARM)
                ?: error("No default alarm sound")
            player.setAudioAttributes(attributes)
            player.setDataSource(this, uri)
            player.prepare()
        }
        if (chosen.isFailure) {
            // Media providers are unavailable before the first unlock; the bundled sound always works.
            Log.w(TAG, "Falling back to bundled alarm sound", chosen.exceptionOrNull())
            player.reset()
            player.setAudioAttributes(attributes)
            resources.openRawResourceFd(R.raw.fallback_alarm).use { afd ->
                player.setDataSource(afd.fileDescriptor, afd.startOffset, afd.length)
            }
            player.prepare()
        }
        player.isLooping = true
        val startVolume = if (settings.volumeRamp) RAMP_START_VOLUME else 1f
        player.setVolume(startVolume, startVolume)
        player.start()
        this.player = player
        if (settings.volumeRamp) {
            rampJob = scope.launch {
                for (step in 1..RAMP_STEPS) {
                    delay(1.seconds)
                    val volume = RAMP_START_VOLUME + (1f - RAMP_START_VOLUME) * step / RAMP_STEPS
                    player.setVolume(volume, volume)
                }
            }
        }
    }

    @Suppress("DEPRECATION")
    private fun startVibration() {
        val effect = VibrationEffect.createWaveform(longArrayOf(0, 800, 600), 0)
        if (Build.VERSION.SDK_INT >= 33) {
            vibrator.vibrate(effect, VibrationAttributes.createForUsage(VibrationAttributes.USAGE_ALARM))
        } else {
            vibrator.vibrate(effect, AudioAttributes.Builder().setUsage(AudioAttributes.USAGE_ALARM).build())
        }
    }

    private fun acquireWakeLock() {
        if (wakeLock?.isHeld == true) return
        wakeLock = getSystemService(PowerManager::class.java)
            .newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "reveille:ringing")
            .apply { acquire(WAKE_LOCK_TIMEOUT_MS) }
    }

    private fun stopOutput() {
        rampJob?.cancel()
        rampJob = null
        player?.run {
            runCatching { stop() }
            release()
        }
        player = null
        vibrator.cancel()
    }

    private fun stopRinging() {
        timeoutJob?.cancel()
        stopOutput()
        current.value = null
        wakeLock?.takeIf { it.isHeld }?.release()
        stopForeground(STOP_FOREGROUND_REMOVE)
        stopSelf()
    }

    override fun onDestroy() {
        stopOutput()
        wakeLock?.takeIf { it.isHeld }?.release()
        scope.cancel()
        super.onDestroy()
    }

    companion object {
        const val ACTION_RING = "com.replydev.reveille.RING"
        const val ACTION_SNOOZE = "com.replydev.reveille.SNOOZE"
        const val ACTION_DISMISS = "com.replydev.reveille.DISMISS"
        private const val EXTRA_ARMED = "armed"
        private const val EXTRA_BACKUP = "backup"
        private const val MAX_AUTO_SNOOZES = 3
        private const val RAMP_STEPS = 30
        private const val RAMP_START_VOLUME = 0.1f
        private const val WAKE_LOCK_TIMEOUT_MS = 40 * 60 * 1000L
        private const val TAG = "Reveille"

        private val current = MutableStateFlow<Ringing?>(null)

        /** The alarm currently ringing, or null. RingingActivity closes itself when this becomes null. */
        val ringing: StateFlow<Ringing?> = current.asStateFlow()

        fun start(context: Context, armed: Armed, backup: Boolean) {
            context.startForegroundService(
                Intent(context, RingingService::class.java)
                    .setAction(ACTION_RING)
                    .putExtra(EXTRA_ARMED, armed.encode())
                    .putExtra(EXTRA_BACKUP, backup),
            )
        }

        fun actionIntent(context: Context, action: String): PendingIntent = PendingIntent.getService(
            context,
            action.hashCode(),
            Intent(context, RingingService::class.java).setAction(action),
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT,
        )

        fun send(context: Context, action: String) {
            context.startService(Intent(context, RingingService::class.java).setAction(action))
        }
    }
}
```

```kotlin
package com.replydev.reveille.alarm

import android.content.Context
import android.os.UserManager
import androidx.work.CoroutineWorker
import androidx.work.ExistingPeriodicWorkPolicy
import androidx.work.PeriodicWorkRequestBuilder
import androidx.work.WorkManager
import androidx.work.WorkerParameters
import com.replydev.reveille.container
import java.util.concurrent.TimeUnit

/** Safety net: re-arms periodically in case the system dropped our alarm. */
class WatchdogWorker(context: Context, params: WorkerParameters) : CoroutineWorker(context, params) {
    override suspend fun doWork(): Result {
        applicationContext.container.scheduler.rearm("watchdog")
        return Result.success()
    }

    companion object {
        fun schedule(context: Context) {
            if (!context.getSystemService(UserManager::class.java).isUserUnlocked) return
            WorkManager.getInstance(context).enqueueUniquePeriodicWork(
                "watchdog",
                ExistingPeriodicWorkPolicy.KEEP,
                PeriodicWorkRequestBuilder<WatchdogWorker>(6, TimeUnit.HOURS).build(),
            )
        }
    }
}
```

```kotlin
import kotlinx.coroutines.launch

/** Re-arms after reboot, app update, clock or time zone change, and exact-alarm permission changes. */
class SystemEventsReceiver : BroadcastReceiver() {
    override fun onReceive(context: Context, intent: Intent) {
        val pending = goAsync()
        CoroutineScope(SupervisorJob() + Dispatchers.IO).launch {
            try {
                context.container.scheduler.rearm(intent.action ?: "system event")
                WatchdogWorker.schedule(context)
            } finally {
                pending.finish()
            }
        }
    }
}
```

```kotlin
    val detail: String,

    val fix: Intent?,

class ReliabilityChecker(private val context: Context, private val settings: SettingsStore) {
    private val packageName = context.packageName
    private val packageUri = Uri.fromParts("package", packageName, null)

    suspend fun checks(): List<ReliabilityCheck> = automaticChecks() + manualChecks(settings.confirmedChecks.first())

    suspend fun allGood(): Boolean = checks().all { it.ok }

    private fun automaticChecks(): List<ReliabilityCheck> {
        val notifications = context.getSystemService(NotificationManager::class.java)
        val alarms = context.getSystemService(AlarmManager::class.java)
        val power = context.getSystemService(PowerManager::class.java)
        val audio = context.getSystemService(AudioManager::class.java)
        val channelImportance = notifications.getNotificationChannel(Notifier.CHANNEL_ALARM)?.importance ?: NotificationManager.IMPORTANCE_NONE
        return buildList {
            add(
                ReliabilityCheck(
                    "notifications", "Notifications allowed",
                    "The ringing alarm and missed-alarm warnings are notifications.",
                    NotificationManagerCompat.from(context).areNotificationsEnabled(), false,
                    Intent(Settings.ACTION_APP_NOTIFICATION_SETTINGS).putExtra(Settings.EXTRA_APP_PACKAGE, packageName),
                ),
            )
            if (Build.VERSION.SDK_INT >= 34) {
                add(
                    ReliabilityCheck(
                        "full_screen", "Full-screen alarms allowed",
                        "Lets the alarm screen appear over the lock screen.",
                        notifications.canUseFullScreenIntent(), false,
                        Intent(Settings.ACTION_MANAGE_APP_USE_FULL_SCREEN_INTENT, packageUri),
                    ),
                )
            }
            add(
                ReliabilityCheck(
                    "alarm_channel", "Alarm notifications at high importance",
                    "If lowered, the alarm screen will not pop up.",
                    channelImportance >= NotificationManager.IMPORTANCE_HIGH, false,
                    Intent(Settings.ACTION_CHANNEL_NOTIFICATION_SETTINGS)
                        .putExtra(Settings.EXTRA_APP_PACKAGE, packageName)
                        .putExtra(Settings.EXTRA_CHANNEL_ID, Notifier.CHANNEL_ALARM),
                ),
            )
            add(
                ReliabilityCheck(
                    "exact_alarms", "Exact alarms allowed",
                    "Required to ring at the exact minute.",
                    alarms.canScheduleExactAlarms(), false,
                    Intent(Settings.ACTION_REQUEST_SCHEDULE_EXACT_ALARM, packageUri),
                ),
            )
            add(
                ReliabilityCheck(
                    "battery_optimization", "Battery optimization off",
                    "Battery-optimized apps can be delayed or stopped in the background.",
                    power.isIgnoringBatteryOptimizations(packageName), false,
                    Intent(Settings.ACTION_REQUEST_IGNORE_BATTERY_OPTIMIZATIONS, packageUri),
                ),
            )
            add(
                ReliabilityCheck(
                    "alarm_volume", "Alarm volume not muted",
                    "Alarms play on the alarm volume, which is separate from the ringer.",
                    audio.getStreamVolume(AudioManager.STREAM_ALARM) > 0, false,
                    Intent(Settings.ACTION_SOUND_SETTINGS),
                ),
            )
        }
    }

    private fun manualChecks(confirmed: Set<String>): List<ReliabilityCheck> {
        fun manual(id: String, title: String, detail: String, fix: Intent?) = ReliabilityCheck(id, title, detail, id in confirmed, true, fix)
        if (!isXiaomi()) {
            return listOf(
                manual(
                    "oem_battery", "Manufacturer battery settings",
                    "Some phones stop apps despite Android's own settings. Open the app's battery settings, choose Unrestricted, then tick Done.",
                    appDetailsIntent(context),
                ),
            )
        }
        return listOf(
            manual(
                "miui_autostart", "Autostart on",
                "Without Autostart, HyperOS does not tell Reveille that the phone restarted, so alarms are lost after a reboot.",
                Intent().setComponent(ComponentName("com.miui.securitycenter", "com.miui.permcenter.autostart.AutoStartManagementActivity")),
            ),
            manual(
                "miui_battery", "Battery saver: No restrictions",
                "In the app's battery saver setting, choose \"No restrictions\".",
                Intent().setComponent(ComponentName("com.miui.powerkeeper", "com.miui.powerkeeper.ui.HiddenAppsConfigActivity"))
                    .putExtra("package_name", packageName)
                    .putExtra("package_label", "Reveille"),
            ),
            manual(
                "miui_permissions", "Lock screen and pop-up permissions",
                "Under Other permissions, allow \"Show on Lock screen\" and \"Display pop-up windows while running in the background\".",
                Intent("miui.intent.action.APP_PERM_EDITOR")
                    .setClassName("com.miui.securitycenter", "com.miui.permcenter.permissions.PermissionsEditorActivity")
                    .putExtra("extra_pkgname", packageName),
            ),
            manual(
                "miui_recents", "Locked in Recents",
                "Open the recent apps view, long-press Reveille and tap the lock icon.",
                null,
            ),
        )
    }

    private fun isXiaomi(): Boolean =
        Build.MANUFACTURER.equals("Xiaomi", ignoreCase = true) || Build.BRAND.lowercase() in setOf("xiaomi", "redmi", "poco")
}
```

```kotlin
        container.repository.skipOccurrence(sources.map { it.key })
    }
```

```kotlin
                onPlus = { update { it.copy(snoozeMinutes = (it.snoozeMinutes + 1).coerceAtMost(MAX_MINUTES)) } },
```

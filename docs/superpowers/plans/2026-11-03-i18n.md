# Reveille Internationalization Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Reveille follows the Android system or per-app language (en, it, es, fr, de), with locale-aware dates and the phone's 12/24-hour clock, and no in-app picker.

**Architecture:** `:engine` stops producing English. It returns small value types (`DayLabel`, `Relative`, `DaySet`), and the app's new `text/` package (`TimeFormats`, `Labels`) turns them into strings via resources. Every user-visible literal moves to `res/values*/strings.xml`. `generateLocaleConfig` exposes the languages to Android 13+'s per-app language setting.

**Tech Stack:** Kotlin, Jetpack Compose, Glance, Android resources (`strings.xml`, `plurals`), `android.text.format.DateFormat`, `java.time`, Robolectric, Maestro.

**Spec:** `docs/superpowers/specs/2026-11-03-i18n-design.md`. Read it first, together with the "one rule" in `CLAUDE.md`.

---

## Ground rules for every task

- Before each commit, run the four checks: `./gradlew :engine:test :app:testDebugUnitTest :app:lintDebug :app:assembleDebug`. All must pass and lint must report 0 errors. Every commit has to build.
- Commit messages are conventional and unsigned, with **no AI attribution or Co-Authored-By trailer**.
- On the alarm path (`alarm/`, `data/`), only string lookups change. Don't touch claim, re-arm or fire-log logic.
- Use `context.resources.configuration.locales[0]` for the locale, never `Locale.getDefault()`.
- `Locale.ENGLISH` / `Locale.US` / `Locale.ROOT` may remain only for machine formats (HTTP, ISO, file names). Task 12 audits this.

### String resource conventions (used by Tasks 5–11)

- File: `app/src/main/res/values/strings.xml`. Keys are snake_case with an area prefix: `nav_`, `today_`, `policy_`, `oneoff_`, `health_`, `settings_`, `update_`, `ringing_`, `notif_`, `widget_`, `time_`, `common_`.
- Use positional args: `<string name="today_skipped">Skipped %1$s %2$s</string>`.
- Use plurals for any count: `<plurals name="today_issues"><item quantity="one">%d issue</item><item quantity="other">%d issues</item></plurals>`.
- Escape apostrophes (`\'`) and quotes (`\"`). Use ` ` for non-breaking spaces. Use `…` (a single character), not `...`.
- In Compose, use `stringResource(R.string.x, args)` / `pluralStringResource(R.plurals.x, n, n)`.
- Outside Compose, use `context.getString(...)` / `context.resources.getQuantityString(...)`.
- Pure helper functions that used to return English take a `Labels` (Task 3) or `Resources` parameter instead of hard-coding text. Their tests become Robolectric (`@RunWith(AndroidJUnit4::class)`) and get `Labels` from `Labels.of(ApplicationProvider.getApplicationContext())`. The English assertions stay identical.
- Log messages, exception messages, `Log.*` tags and machine strings stay as literals.
- Content descriptions are user-visible, so they become resources too.

```kotlin
sealed interface DayLabel {
    data object Today : DayLabel
    data object Tomorrow : DayLabel
    /** Within the next 7 days: rendered as a weekday name. */
    data class Weekday(val date: LocalDate) : DayLabel
    data class Date(val date: LocalDate) : DayLabel
}

/** How far ahead something is, rounded for display. */
sealed interface Relative {
    data object Now : Relative
    data class Minutes(val count: Long) : Relative
    data class Hours(val count: Long) : Relative
    data class Days(val count: Long) : Relative
}

/** A rule's weekdays, summarised. [Custom] days are in Monday-first order. */
sealed interface DaySet {
    data object EveryDay : DaySet
    data object Weekdays : DaySet
    data class Custom(val days: List<DayOfWeek>) : DaySet
}
```

```kotlin
        val today = now.atZone(zone).toLocalDate()

        return when {

            minutes < 60 -> Relative.Minutes(minutes)
            minutes < 48 * 60 -> Relative.Hours((minutes + 30) / 60)
            else -> Relative.Days((minutes + 12 * 60) / (24 * 60))
        }
    }
```

```kotlin
package com.replydev.reveille.text

import android.content.Context
import android.text.format.DateFormat
import java.text.NumberFormat
import java.time.DayOfWeek
import java.time.LocalDate
import java.time.LocalTime
import java.time.format.DateTimeFormatter
import java.time.format.TextStyle
import java.util.Locale

/** A time split for large displays: "7:30" and "PM" are drawn in different sizes. [period] is null in 24-hour mode. */
data class TimeParts(val digits: String, val period: String?)

/**
 * Locale-aware formatting for everything Reveille shows about time. Date patterns come from the locale's own
 * skeletons (CLDR), so word order and punctuation are native ("lunedì 2 novembre", "Montag, 2. November").
 */
class TimeFormats(val locale: Locale, val is24Hour: Boolean) {
    private fun skeleton(skeleton: String) = DateTimeFormatter.ofPattern(DateFormat.getBestDateTimePattern(locale, skeleton), locale)

    private val timeFormat = if (is24Hour) DateTimeFormatter.ofPattern("HH:mm", locale) else skeleton("hmm")
    private val digitsFormat = if (is24Hour) timeFormat else DateTimeFormatter.ofPattern("h:mm", locale)
    private val periodFormat = DateTimeFormatter.ofPattern("a", locale)
    private val headerFormat = skeleton("EEEEdMMMM")
    private val shortDateFormat = skeleton("EEEdMMM")
    private val shortDateYearFormat = skeleton("EEEdMMMyyyy")
    private val dayMonthFormat = skeleton("dMMM")
    private val fullDateFormat = skeleton("dMMMyyyy")
    private val sizeFormat = NumberFormat.getNumberInstance(locale).apply {
        minimumFractionDigits = 1
        maximumFractionDigits = 1
    }

    fun time(value: LocalTime): String = value.format(timeFormat)

    fun timeParts(value: LocalTime): TimeParts =
        TimeParts(value.format(digitsFormat), if (is24Hour) null else value.format(periodFormat))

    /** "Monday, 2 November": Today header and ringing screen. */
    fun headerDate(value: LocalDate): String = value.format(headerFormat)

    /** "Mon 2 Nov", or with the year when it isn't [currentYear]. */
    fun shortDate(value: LocalDate, currentYear: Int = value.year): String =
        value.format(if (value.year == currentYear) shortDateFormat else shortDateYearFormat)

    /** "2 Nov". */
    fun dayMonth(value: LocalDate): String = value.format(dayMonthFormat)

    /** "2 Nov 2026". */
    fun fullDate(value: LocalDate): String = value.format(fullDateFormat)

    fun weekdayShort(day: DayOfWeek): String = day.getDisplayName(TextStyle.SHORT, locale)
    fun weekdayNarrow(day: DayOfWeek): String = day.getDisplayName(TextStyle.NARROW, locale)
    fun weekdayFull(day: DayOfWeek): String = day.getDisplayName(TextStyle.FULL, locale)

    /** Decimal megabytes with one decimal, as Android's own storage screens count them. */
    fun size(bytes: Long): String = "${sizeFormat.format(bytes / 1_000_000.0)} MB"

    companion object {
        fun of(context: Context): TimeFormats =
            TimeFormats(context.resources.configuration.locales[0], DateFormat.is24HourFormat(context))
    }
}
```

```kotlin
package com.replydev.reveille.text

import android.content.Context
import android.content.res.Resources
import com.replydev.reveille.R
import com.replydev.reveille.engine.DayLabel
import com.replydev.reveille.engine.DaySet
import com.replydev.reveille.engine.Relative
import com.replydev.reveille.engine.TimeText
import java.time.Instant
import java.time.LocalDate
import java.time.ZoneId

/** Turns the engine's time values into words in the current language. The only place that does. */
class Labels(private val res: Resources, val formats: TimeFormats) {
    fun day(label: DayLabel): String = when (label) {
        DayLabel.Today -> res.getString(R.string.time_today)
        DayLabel.Tomorrow -> res.getString(R.string.time_tomorrow)
        is DayLabel.Weekday -> formats.weekdayShort(label.date.dayOfWeek)
        is DayLabel.Date -> formats.shortDate(label.date)
    }

    fun relative(value: Relative): String = when (value) {
        Relative.Now -> res.getString(R.string.time_now)
        is Relative.Minutes -> plural(R.plurals.time_in_minutes, value.count)
        is Relative.Hours -> plural(R.plurals.time_in_hours, value.count)
        is Relative.Days -> plural(R.plurals.time_in_days, value.count)
    }

    fun relative(at: Instant, now: Instant): String = relative(TimeText.relativeOf(at, now))

    fun days(set: DaySet): String = when (set) {
        DaySet.EveryDay -> res.getString(R.string.time_every_day)
        DaySet.Weekdays -> res.getString(R.string.time_weekdays)
        is DaySet.Custom -> set.days.joinToString(" ") { formats.weekdayShort(it) }
    }

    /** "Tomorrow 07:55", "Thu 08:55", "Wed 18 Nov 08:00". */
    fun dayAndTime(at: Instant, now: Instant, zone: ZoneId): String =
        res.getString(R.string.time_day_and_time, dayPart(at, now, zone), formats.time(at.atZone(zone).toLocalTime()))

    /** The day half of [dayAndTime]. */
    fun dayPart(at: Instant, now: Instant, zone: ZoneId): String = day(TimeText.dayOf(at, now, zone))
```

```kotlin
@RunWith(AndroidJUnit4::class)
class TranslationsTest {
    private val context: Context get() = ApplicationProvider.getApplicationContext()

    @Test @Config(qualifiers = "it")
    fun italian() {
        assertEquals("Oggi", context.getString(R.string.nav_today))
        assertEquals("Sveglia", context.getString(R.string.oneoff_default_label))
        assertEquals("tra 1 min", Labels.of(context).relative(Relative.Minutes(1)))
        assertEquals("Domani", Labels.of(context).day(DayLabel.Tomorrow))

    @Test @Config(qualifiers = "de")
    fun german() = assertEquals("Heute", context.getString(R.string.nav_today))

    @Test @Config(qualifiers = "es")
    fun spanish() = assertEquals("Hoy", context.getString(R.string.nav_today))

    @Test @Config(qualifiers = "fr")
    fun french() = assertEquals("Aujourd’hui", context.getString(R.string.nav_today))
}
```
